## 为什么使用三个位图

每节电池需要保留故障位置，而整组可能同时发生多个问题。一个字节的不同位可以分别表示不同单体：

| 位图值 | 低三位 | 对应单体 |
| --- | --- | --- |
| 0x00 | 000 | 没有记录这类问题 |
| 0x01 | 001 | 第 1 节 |
| 0x02 | 010 | 第 2 节 |
| 0x04 | 100 | 第 3 节 |
| 0x05 | 101 | 第 1、3 节 |
| 0x07 | 111 | 第 1、2、3 节 |

从右往左，最低位是 bit0。数组下标 0、1、2 分别对应 bit0、bit1、bit2。

分别使用 over_voltage_mask、invalid_mask、timeout_mask，便于后续日志和 CAN 同时表达故障类型与位置。每个位图当前使用 uint8_t，最多表示 8 节；本项目宏为 3，计划的 5 串也在这一范围内。若将来超过 8 节，必须修改位图类型和测试。

额外的 input_error 用于整组指针或参数错误。此时三个位图为零仍不能视为检查正常，调用者必须检查所有四个字段。

## 第一步：在 bms_protection.h 添加类型和声明

打开 [bms_protection.h](../MCU/projects/bms_bringup/Core/Inc/bms_protection.h)，保留已有内容，将下方代码放在最后的 #endif 之前：

```c
/*
 * 三节单体电压的检查汇总。
 * 三个位图均约定：bit0 对应第1节，bit1 对应第2节，bit2 对应第3节。
 * 对应位为 1，表示该节存在这一类问题。uint8_t 最多容纳 8 节。
 */
typedef struct
{
    uint8_t over_voltage_mask; /* 有效且未超时的数据中，哪些单体达到过压阈值 */
    uint8_t invalid_mask;      /* 哪些单体没有可用数据，或电压数值不合理 */
    uint8_t timeout_mask;      /* 哪些单体的数据已经超时 */
    uint8_t input_error;       /* 1：整组指针或检查参数错误；0：参数检查通过 */
} BmsVoltageReport;

/*
 * 汇总单体电压的过压、无效和超时情况。
 * sample：整组采样快照；这里只读取 cell_mv 数组。
 * ov_threshold_mv：单体过压阈值，单位 mV，必须大于 0。
 * now_ms、max_age_ms：与前面的时效检查使用相同含义。
 * 返回值是本次检查结果；这里不保存故障历史，也不控制负载。
 */
BmsVoltageReport Bms_CheckPackVoltages(
    const BmsSample *sample,
    int32_t ov_threshold_mv,
    uint32_t now_ms,
    uint32_t max_age_ms);
```

BmsVoltageReport 是一个结构体，四个字段作为一个整体返回，不需要动态申请内存。此结构体表示本次检查结果，不记录过去的故障。

## 第二步：在 bms_protection.c 添加实现

打开 [bms_protection.c](../MCU/projects/bms_bringup/Core/Src/bms_protection.c)，将下面函数追加到现有函数后面。文件顶部已经包含 bms_protection.h 和 stddef.h，后者提供 NULL。

```c
BmsVoltageReport Bms_CheckPackVoltages(
    const BmsSample *sample,
    int32_t ov_threshold_mv,
    uint32_t now_ms,
    uint32_t max_age_ms)
{
    /* 每次从全零结果开始，避免把上一次检查结果带进来。 */
    BmsVoltageReport report = {0};
    uint32_t i;
    uint8_t cell_bit;
    const BmsMeasurement *cell;
    BmsSampleStatus sample_status;
    BmsOvResult ov_result;

    /* 参数错误单独报告，调用者必须同时检查 input_error。 */
    if (sample == NULL || ov_threshold_mv <= 0 || max_age_ms == 0u)
    {
        report.input_error = 1u;
        return report;
    }

    for (i = 0u; i < BMS_CELL_COUNT; i++)
    {
        cell = &sample->cell_mv[i];

        /* i 为 0、1、2 时，分别得到二进制 001、010、100。 */
        cell_bit = (uint8_t)(1u << i);

        /* 先确认这一路数据可用且没有过期。 */
        sample_status = Bms_CheckSampleFreshness(cell, now_ms, max_age_ms);

        if (sample_status == BMS_SAMPLE_INVALID)
        {
            report.invalid_mask |= cell_bit;
            continue; /* 这一节不能继续检查，转到下一节。 */
        }

        if (sample_status == BMS_SAMPLE_TIMEOUT)
        {
            report.timeout_mask |= cell_bit;
            continue; /* 旧数据不用于报告当前过压。 */
        }

        /* 只有通过时效检查的数据，才继续检查电压数值。 */
        ov_result = Bms_CheckCellOverVoltage(cell, ov_threshold_mv);

        if (ov_result == BMS_OV_DATA_INVALID)
        {
            report.invalid_mask |= cell_bit;
        }
        else if (ov_result == BMS_OV_TRIGGERED)
        {
            /* 按位或可以保留前面已经记录的其他单体。 */
            report.over_voltage_mask |= cell_bit;
        }
    }

    return report;
}
```

关键语句的含义：

| 语句或表达式 | 含义 |
| --- | --- |
| report = {0} | 创建本次结果，并将所有字段初始化为零 |
| &sample->cell_mv[i] | 取得第 i 个单体测量对象的地址，交给已有函数读取 |
| 1u << i | 将最低位的 1 左移 i 位，得到本节对应的位置 |
| mask |= cell_bit | 把本节对应位置为 1，同时保留其他单体已经置位的结果 |
| continue | 本节已记录无效或超时，直接检查下一节 |
| return report | 返回整份检查结果，由调用者决定如何记录或处理 |

检查顺序是：整组参数 → 单路标志与来源 → 单路超时 → 电压数值与过压。

同一节本次按此顺序归入一类结果。例如，旧数据即使是 4300 mV，也只报告超时；全零快照因 valid=0 和 source=NONE 报告无效。不同节的问题可以同时记录，不能发现第一处问题就提前结束循环。

这是当前快照的分类规则，后续状态机中的故障历史和锁存需要单独维护。

## 第三步：新建电脑端测试文件

在 MCU/projects/bms_bringup/Core/tests 下新建 test_bms_pack_voltage.c，填写：

```c
#include <assert.h>
#include <stdio.h>
#include "bms_protection.h"

/* 每个场景从同一份正常快照开始，避免受上一场景影响。 */
static BmsSample MakeNormalSample(void)
{
    BmsSample sample = {0};
    uint32_t i;

    for (i = 0u; i < BMS_CELL_COUNT; i++)
    {
        sample.cell_mv[i].value = 3700;
        sample.cell_mv[i].timestamp_ms = 1000u;
        sample.cell_mv[i].valid = 1u;
        sample.cell_mv[i].source = BMS_SOURCE_SIMULATED;
    }

    return sample;
}

/* 每个场景都核对四个字段，避免遗漏意外产生的其他故障。 */
static void AssertReport(
    BmsVoltageReport report,
    uint8_t expected_ov,
    uint8_t expected_invalid,
    uint8_t expected_timeout,
    uint8_t expected_input_error)
{
    assert(report.over_voltage_mask == expected_ov);
    assert(report.invalid_mask == expected_invalid);
    assert(report.timeout_mask == expected_timeout);
    assert(report.input_error == expected_input_error);
}

int main(void)
{
    BmsSample sample;
    BmsSample empty_sample = {0}; /* 模拟上电后尚未取得采样的快照。 */
    uint32_t i;

    /* 这些是软件测试参数，尚不是实物电池的保护配置。 */
    const int32_t ov_threshold_mv = 4200;
    const uint32_t now_ms = 1499u;
    const uint32_t max_age_ms = 500u;

    /* 1：三节均有效、未超时，且未达到过压阈值。 */
    sample = MakeNormalSample();
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0u, 0u, 0u);

    /* 2：第1节刚好达到阈值，bit0 应为 1。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4200;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x01u, 0u, 0u, 0u);

    /* 3：第2节过压，bit1 应为 1。 */
    sample = MakeNormalSample();
    sample.cell_mv[1].value = 4300;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x02u, 0u, 0u, 0u);

    /* 4：第3节过压，bit2 应为 1。 */
    sample = MakeNormalSample();
    sample.cell_mv[2].value = 4300;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x04u, 0u, 0u, 0u);

    /* 5：第1、3节同时过压，001 | 100 = 101，即十六进制 0x05。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4300;
    sample.cell_mv[2].value = 4300;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x05u, 0u, 0u, 0u);

    /* 6：无效标志、无来源、负电压，分别出现在三节上。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].valid = 0u;
    sample.cell_mv[1].source = BMS_SOURCE_NONE;
    sample.cell_mv[2].value = -1;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0x07u, 0u, 0u);

    /* 7：同时出现过压、无效和超时，不能在发现第一项后提前结束。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4300;
    sample.cell_mv[1].valid = 0u;
    sample.cell_mv[2].timestamp_ms = 999u; /* 数据年龄刚好 500 ms。 */
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x01u, 0x02u, 0x04u, 0u);

    /* 8：第2节旧值虽然高于阈值，本次只报告它超时。 */
    sample = MakeNormalSample();
    sample.cell_mv[1].value = 4300;
    sample.cell_mv[1].timestamp_ms = 999u;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0u, 0x02u, 0u);

    /* 9：没有整组采样对象，必须报告输入错误。 */
    AssertReport(Bms_CheckPackVoltages(
        NULL, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0u, 0u, 1u);

    /* 10：过压阈值为 0，不能当作正常检查结果。 */
    sample = MakeNormalSample();
    AssertReport(Bms_CheckPackVoltages(
        &sample, 0, now_ms, max_age_ms),
        0u, 0u, 0u, 1u);

    /* 11：超时参数为 0，也必须报告输入错误。 */
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, 0u),
        0u, 0u, 0u, 1u);

    /* 12：计数回绕后，三节数据年龄都刚好为 500 ms。 */
    sample = MakeNormalSample();
    for (i = 0u; i < BMS_CELL_COUNT; i++)
    {
        sample.cell_mv[i].timestamp_ms = UINT32_MAX - 99u;
    }
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, 400u, max_age_ms),
        0u, 0u, 0x07u, 0u);

    /* 13：全零快照代表尚未采集，三节都应报告数据无效。 */
    sample = empty_sample;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0x07u, 0u, 0u);

    puts("13 pack voltage scenarios passed.");
    return 0;
}
```

MakeNormalSample() 负责构造软件快照，每个场景重新取得一份，避免前一场景对后一场景产生影响。函数名中的 Make 表示构造，不代表硬件采样。

AssertReport() 每次比较四个字段，因此这里是 13 个测试场景、52 项断言。打印的数量是手写说明，不是测试框架自动统计。

测试的 main() 是 Windows 程序入口，不能加入 Keil 的 MCU 目标。现有 bms_protection.c 已在 Keil 工程中，本步没有新 MCU 源文件需要添加。
