## 本步目标与调用关系

把 Bms_CheckPackVoltages()、Bms_CheckTemperature() 和 Bms_CheckCurrent() 的返回结果交给 Bms_CombineFaults()，得到 uint32_t 故障位图。后续状态机再读取该位图决定是否允许运行。

每轮必须先完成三类检查，再汇总同一轮结果。不能只把电压报告清零、手填两个 NORMAL，就当作采样通过；本函数只转换已知结果，不重新检查原始采样和时间戳。

| 故障位 | 含义 | 十六进制值 |
| --- | --- | --- |
| bit0 | 至少一节过压 | 0x01 |
| bit1 | 至少一节欠压 | 0x02 |
| bit2 | 过温 | 0x04 |
| bit3 | 放电过流 | 0x08 |
| bit4 | 充电过流 | 0x10 |
| bit5 | 至少一路数据无效 | 0x20 |
| bit6 | 至少一路数据超时 | 0x40 |
| bit7 | 输入参数或结果类型错误 | 0x80 |

原电压报告的位表示“哪节电池”；新故障位图的位表示“哪类故障”。例如第3节过压：原 over_voltage_mask 为 0x04，汇总后 CELL_OV 为 0x01。仍需保留原电压报告用于定位单体；总位图不记录无效或超时的具体通道。

0 表示本次已执行的检查没有发现故障，不等于完成全系统自检，也不意味着可以绕过状态机开启负载。该函数不保留历史；故障锁定、清除条件与命令留到状态机实现。位号是当前软件约定，CAN 报文格式尚未冻结。

## 第一步：增加故障类型和声明

在 Core/Inc/bms_protection.h 最后的 #endif 前添加，放在现有温度、电流类型定义之后：

```c
/*
 * 每一位代表一种故障，可以同时记录多种故障。
 * 注意：这里的位表示故障类型，不表示第几节电池。
 */
typedef enum
{
    BMS_FAULT_NONE           = 0u,       /* 本次没有发现已检查的故障 */
    BMS_FAULT_CELL_OV        = 1u << 0,  /* 单体过压 */
    BMS_FAULT_CELL_UV        = 1u << 1,  /* 单体欠压 */
    BMS_FAULT_OVER_TEMP      = 1u << 2,  /* 过温 */
    BMS_FAULT_DISCHARGE_OC   = 1u << 3,  /* 放电过流 */
    BMS_FAULT_CHARGE_OC      = 1u << 4,  /* 充电过流 */
    BMS_FAULT_DATA_INVALID   = 1u << 5,  /* 至少一路数据无效 */
    BMS_FAULT_SAMPLE_TIMEOUT = 1u << 6,  /* 至少一路数据超时 */
    BMS_FAULT_INPUT_ERROR    = 1u << 7   /* 检查参数或结果类型错误 */
} BmsFaultFlag;

/*
 * 汇总已经得到的电压、温度和电流检查结果。
 * 返回值是故障位图，使用 uint32_t 保存多个故障位。
 * 只描述本次结果；历史故障的锁定由后续状态机负责。
 */
uint32_t Bms_CombineFaults(
    const BmsVoltageReport *voltage,
    BmsTemperatureResult temperature_result,
    BmsCurrentResult current_result);
```

1u << n 把无符号整数 1 左移 n 位，使每个故障拥有不同的二进制位。BmsFaultFlag 用于命名单个位，组合后的值使用 uint32_t 保存。

## 第二步：添加汇总函数

在 Core/Src/bms_protection.c 的现有函数后添加。该文件已有 NULL 所需的标准头文件：

```c
uint32_t Bms_CombineFaults(
    const BmsVoltageReport *voltage,
    BmsTemperatureResult temperature_result,
    BmsCurrentResult current_result)
{
    /* 每次从零开始，避免上次结果混入本次检查。 */
    uint32_t faults = BMS_FAULT_NONE;

    if (voltage == NULL)
    {
        faults |= BMS_FAULT_INPUT_ERROR;
        /* 仍继续汇总温度和电流，保留其他已知故障。 */
    }
    else
    {
        /* 独立判断各字段，因为不同电池可能同时存在不同问题。 */
        if (voltage->over_voltage_mask != 0u)
        {
            faults |= BMS_FAULT_CELL_OV;
        }

        if (voltage->under_voltage_mask != 0u)
        {
            faults |= BMS_FAULT_CELL_UV;
        }

        if (voltage->invalid_mask != 0u)
        {
            faults |= BMS_FAULT_DATA_INVALID;
        }

        if (voltage->timeout_mask != 0u)
        {
            faults |= BMS_FAULT_SAMPLE_TIMEOUT;
        }

        if (voltage->input_error != 0u)
        {
            faults |= BMS_FAULT_INPUT_ERROR;
        }
    }

    switch (temperature_result)
    {
        case BMS_TEMP_NORMAL:
            break;

        case BMS_TEMP_OVER_LIMIT:
            faults |= BMS_FAULT_OVER_TEMP;
            break;

        case BMS_TEMP_DATA_INVALID:
            faults |= BMS_FAULT_DATA_INVALID;
            break;

        case BMS_TEMP_TIMEOUT:
            faults |= BMS_FAULT_SAMPLE_TIMEOUT;
            break;

        case BMS_TEMP_INPUT_ERROR:
        default:
            /* 未定义的枚举值也作为输入错误，不能当作正常。 */
            faults |= BMS_FAULT_INPUT_ERROR;
            break;
    }

    switch (current_result)
    {
        case BMS_CURRENT_NORMAL:
            break;

        case BMS_CURRENT_DISCHARGE_OVER_LIMIT:
            faults |= BMS_FAULT_DISCHARGE_OC;
            break;

        case BMS_CURRENT_CHARGE_OVER_LIMIT:
            faults |= BMS_FAULT_CHARGE_OC;
            break;

        case BMS_CURRENT_DATA_INVALID:
            faults |= BMS_FAULT_DATA_INVALID;
            break;

        case BMS_CURRENT_TIMEOUT:
            faults |= BMS_FAULT_SAMPLE_TIMEOUT;
            break;

        case BMS_CURRENT_INPUT_ERROR:
        default:
            faults |= BMS_FAULT_INPUT_ERROR;
            break;
    }

    return faults;
}
```

faults |= flag 等价于 faults = faults | flag，增加故障位并保留已经发现的其他故障。电压字段使用独立的 if，因为不同单体可以同时出现不同问题；温度和电流分别用 switch 映射单个结果。

电压报告为空时先记录输入错误，继续收集另外两类结果，避免漏掉已知故障。default 接住未定义的枚举值，将其报告为输入错误。

判断“是否存在过温”，使用 (faults & BMS_FAULT_OVER_TEMP) != 0u。只有判断“是否全部无故障”才使用 faults == BMS_FAULT_NONE；faults == BMS_FAULT_OVER_TEMP 会漏掉过温与其他故障并存的情况。

## 第三步：新建测试

新建 Core/tests/test_bms_faults.c：

```c
#include <assert.h>
#include <stdio.h>
#include "bms_protection.h"

int main(void)
{
    BmsVoltageReport voltage = {0};
    const BmsVoltageReport empty_report = {0};

    /* 1：三类检查都没有发现故障。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_NONE);

    /* 2：第1节过压，汇总成“存在单体过压”。 */
    voltage.over_voltage_mask = 0x01u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_CELL_OV);
    voltage = empty_report;

    /* 3：第2节欠压。 */
    voltage.under_voltage_mask = 0x02u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_CELL_UV);
    voltage = empty_report;

    /* 4：单独过温。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_OVER_LIMIT, BMS_CURRENT_NORMAL)
        == BMS_FAULT_OVER_TEMP);

    /* 5：单独放电过流。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_DISCHARGE_OVER_LIMIT)
        == BMS_FAULT_DISCHARGE_OC);

    /* 6：单独充电过流。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_CHARGE_OVER_LIMIT)
        == BMS_FAULT_CHARGE_OC);

    /* 7：第3节电压数据无效。 */
    voltage.invalid_mask = 0x04u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_DATA_INVALID);
    voltage = empty_report;

    /* 8：温度数据无效。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_DATA_INVALID, BMS_CURRENT_NORMAL)
        == BMS_FAULT_DATA_INVALID);

    /* 9：电流数据无效。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_DATA_INVALID)
        == BMS_FAULT_DATA_INVALID);

    /* 10：第1节电压数据超时。 */
    voltage.timeout_mask = 0x01u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_SAMPLE_TIMEOUT);
    voltage = empty_report;

    /* 11：温度超时。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_TIMEOUT, BMS_CURRENT_NORMAL)
        == BMS_FAULT_SAMPLE_TIMEOUT);

    /* 12：电流超时。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_TIMEOUT)
        == BMS_FAULT_SAMPLE_TIMEOUT);

    /* 13：电压检查报告参数错误。 */
    voltage.input_error = 1u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_INPUT_ERROR);
    voltage = empty_report;

    /* 14：温度检查报告参数错误。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_INPUT_ERROR, BMS_CURRENT_NORMAL)
        == BMS_FAULT_INPUT_ERROR);

    /* 15：电流检查报告参数错误。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_INPUT_ERROR)
        == BMS_FAULT_INPUT_ERROR);

    /* 16：没有电压报告，不能认为全部正常。 */
    assert(Bms_CombineFaults(
        NULL, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_INPUT_ERROR);

    /*
     * 17：不同通道同时故障，所有对应位都要保留。
     * 第1节过压、第2节欠压、第3节无效；
     * 温度超时，同时放电过流。
     */
    voltage.over_voltage_mask = 0x01u;
    voltage.under_voltage_mask = 0x02u;
    voltage.invalid_mask = 0x04u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_TIMEOUT, BMS_CURRENT_DISCHARGE_OVER_LIMIT)
        == (BMS_FAULT_CELL_OV |
            BMS_FAULT_CELL_UV |
            BMS_FAULT_DATA_INVALID |
            BMS_FAULT_SAMPLE_TIMEOUT |
            BMS_FAULT_DISCHARGE_OC));

    /* 18：电压报告为空时，仍保留过温和充电过流。 */
    assert(Bms_CombineFaults(
        NULL, BMS_TEMP_OVER_LIMIT, BMS_CURRENT_CHARGE_OVER_LIMIT)
        == (BMS_FAULT_INPUT_ERROR |
            BMS_FAULT_OVER_TEMP |
            BMS_FAULT_CHARGE_OC));

    /* 19：新的检查全部正常，汇总函数不保留历史故障。 */
    voltage = empty_report;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_NONE);

    /* 20：传入未定义的温度结果，报告输入错误。 */
    assert(Bms_CombineFaults(
        &voltage, (BmsTemperatureResult)99, BMS_CURRENT_NORMAL)
        == BMS_FAULT_INPUT_ERROR);

    /* 21：传入未定义的电流结果，报告输入错误。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, (BmsCurrentResult)99)
        == BMS_FAULT_INPUT_ERROR);

    puts("21 fault aggregation tests passed.");
    return 0;
}
```

21 项断言覆盖正常、所有单项映射、三类数据无效与超时、输入错误、空指针、多故障并存、连续调用不保留历史及未知枚举值。测试直接构造检查结果，目的是验证汇总映射；原有六组测试继续验证测量检查逻辑。

empty_report 是字段为零的结构体。voltage = empty_report 是结构体赋值，重置全部字段，让后一个场景不受前一个场景的数据影响。这里手填正常结果仅作为测试输入，实际系统必须调用前置检查函数取得结果。
