## 第一步：替换返回结构体

在 Core/Inc/bms_protection.h 中，将原 BmsVoltageReport 的注释和结构体完整替换为：

```c
/*
 * 四个位图分别记录过压、欠压、无效和超时位置。
 * bit0、bit1、bit2 对应第1、2、3节；uint8_t 最多表示8节。
 */
typedef struct
{
    uint8_t over_voltage_mask;  /* 哪些单体达到过压阈值 */
    uint8_t under_voltage_mask; /* 哪些单体达到欠压阈值 */
    uint8_t invalid_mask;       /* 哪些单体的数据无效 */
    uint8_t timeout_mask;       /* 哪些单体的数据超时 */
    uint8_t input_error;        /* 1：整组指针或参数错误 */
} BmsVoltageReport;
```

位图位置仍为 bit0 对应第1节、bit1 对应第2节、bit2 对应第3节。过压与欠压使用不同位图，因此第1节欠压和第2节过压可以同时表达。

## 第二步：替换汇总函数声明

在同一头文件中，将原 Bms_CheckPackVoltages 的注释和声明替换为：

```c
/*
 * 检查各节电池的电压上下限和数据时效。
 * 所有电压参数单位为 mV，时间参数单位为 ms。
 * 参数必须满足：0 < uv_threshold_mv < ov_threshold_mv。
 * 返回本次检查结果，不记录历史故障，也不直接控制负载。
 */
BmsVoltageReport Bms_CheckPackVoltages(

    int32_t ov_threshold_mv,

    uint32_t max_age_ms);
```

其他枚举和单节检查声明保留。新增的欠压参数放在过压参数前面，固定顺序为下限、上限、当前时间、最大数据年龄。

## 第三步：替换汇总函数实现

在 Core/Src/bms_protection.c 中，完整替换原 Bms_CheckPackVoltages，保留另外三个单路函数：

```c
BmsVoltageReport Bms_CheckPackVoltages(
    const BmsSample *sample,
    int32_t uv_threshold_mv,
    int32_t ov_threshold_mv,
    uint32_t now_ms,
    uint32_t max_age_ms)
{
    BmsVoltageReport report = {0};
    uint32_t i;
    uint8_t cell_bit;
    const BmsMeasurement *cell;
    BmsSampleStatus sample_status;
    BmsOvResult ov_result;
    BmsUvResult uv_result;

    /*
     * 下限必须大于0，上限必须高于下限。
     * 上下限相等或顺序颠倒时，不能继续做正常保护判断。
     */
    if (sample == NULL ||
        uv_threshold_mv <= 0 ||
        ov_threshold_mv <= uv_threshold_mv ||
        max_age_ms == 0u)
    {
        report.input_error = 1u;
        return report;
    }

    for (i = 0u; i < BMS_CELL_COUNT; i++)
    {
        cell = &sample->cell_mv[i];
        cell_bit = (uint8_t)(1u << i);

        /* 先检查标志、来源和时间，旧数据不用于报告当前过压或欠压。 */
        sample_status = Bms_CheckSampleFreshness(
            cell, now_ms, max_age_ms);

        if (sample_status == BMS_SAMPLE_INVALID)
        {
            report.invalid_mask |= cell_bit;
            continue;
        }

        if (sample_status == BMS_SAMPLE_TIMEOUT)
        {
            report.timeout_mask |= cell_bit;
            continue;
        }

        /* 复用两个单节函数，不在汇总函数中重复实现阈值比较。 */
        ov_result = Bms_CheckCellOverVoltage(cell, ov_threshold_mv);
        uv_result = Bms_CheckCellUnderVoltage(cell, uv_threshold_mv);

        if (ov_result == BMS_OV_DATA_INVALID ||
            uv_result == BMS_UV_DATA_INVALID)
        {
            report.invalid_mask |= cell_bit;
        }
        else if (ov_result == BMS_OV_TRIGGERED)
        {
            report.over_voltage_mask |= cell_bit;
        }
        else if (uv_result == BMS_UV_TRIGGERED)
        {
            report.under_voltage_mask |= cell_bit;
        }
    }

    return report;
}
```

检查顺序为：整组参数 → 单路数据标志和来源 → 单路时效 → 电压数值和上下限。

参数不合理时单独设置 input_error。有效且未超时的数据才会进入过压与欠压判断；负电压返回无效，标记有效的零电压触发欠压。

上下限必须严格有序。这样同一节有效电压不可能同时达到过压和欠压条件，分支可以按过压、欠压顺序分类。不同节可以分别出现不同问题。

报告每次从零构造，表示本次快照结果；历史故障锁存仍由后续状态机负责。

#include <assert.h>

#include "bms_protection.h"

/* 每个场景重新构造正常快照，避免混入上一场景的数据。 */
static BmsSample MakeNormalSample(void)
{
    BmsSample sample = {0};
    uint32_t i;

    for (i = 0u; i < BMS_CELL_COUNT; i++)
    {
        sample.cell_mv[i].value = 3700;
        sample.cell_mv[i].timestamp_ms = 1000u;
        sample.cell_mv[i].valid = 1u;
        sample.cell_mv[i].source = BMS_SOURCE_SIMULATED;
    }

    return sample;
}

/* 每个场景核对全部五个字段，参数顺序为过压、欠压、无效、超时、输入错误。 */
static void AssertReport(
    BmsVoltageReport report,
    uint8_t expected_ov,
    uint8_t expected_uv,
    uint8_t expected_invalid,
    uint8_t expected_timeout,
    uint8_t expected_input_error)
{
    assert(report.over_voltage_mask == expected_ov);
    assert(report.under_voltage_mask == expected_uv);
    assert(report.invalid_mask == expected_invalid);
    assert(report.timeout_mask == expected_timeout);
    assert(report.input_error == expected_input_error);
}

int main(void)
{
    BmsSample sample;
    BmsSample empty_sample = {0};
    uint32_t i;

    /* 测试用上下限和时间参数，尚不是实物电池的保护配置。 */
    const int32_t uv_threshold_mv = 3000;
    const int32_t ov_threshold_mv = 4200;
    const uint32_t now_ms = 1499u;
    const uint32_t max_age_ms = 500u;

    /* 1：三节数据均正常，五个字段都应为0。 */
    sample = MakeNormalSample();
    AssertReport(Bms_CheckPackVoltages(

        0u, 0u, 0u, 0u, 0u);

    /* 2：第1节刚好达到过压阈值。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4200;

    sample.cell_mv[1].value = 4300;

    /* 5：第1、3节同时过压。 */

    sample.cell_mv[2].timestamp_ms = 999u;

    /* 8：第2节旧值偏高，只报告超时。 */

    /* 15：第2节欠压。 */

    sample.cell_mv[1].value = 2900;
    AssertReport(Bms_CheckPackVoltages(

        0u, 0x02u, 0u, 0u, 0u);

    /* 16：第3节欠压。 */
    sample = MakeNormalSample();
    sample.cell_mv[2].value = 2900;
    AssertReport(Bms_CheckPackVoltages(

    sample.cell_mv[0].value = 2900;
    sample.cell_mv[2].value = 2900;
    AssertReport(Bms_CheckPackVoltages(

    /* 18：第1节欠压、第2节过压，可以同时记录。 */

    sample.cell_mv[0].value = 2900;

## 常见检查点

以下是可能的排查方向，不代表已经发生的问题：

- too few arguments：某处调用仍使用旧参数列表，检查声明、定义和调用是否一致。
- 重复定义：确认结构体和同名函数采用完整替换，而不是保留两份定义。
- expected_uv 相关断言失败：先看位图位置和 AssertReport 参数顺序。
- 单独运行仍输出 13 个场景：确认已保存测试文件且新编译成功，避免运行旧程序。
- 旧值偏低被记录为欠压：确认超时分支通过 continue 跳过了数值检查。
