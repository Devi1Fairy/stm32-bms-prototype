## 统一回归命令及含义

以下命令在仓库根目录由用户按需执行。每个测试单独生成 Windows 可执行文件；它们调用同一份纯 C 业务源码，不依赖板卡和 HAL。构建输出位于已忽略的 build/host。

```powershell
Set-Location -LiteralPath 'D:\嵌入式学习\电池管理系统'
$nativeCompiler = 'E:\mingw\bin\gcc.exe'
$coreDirectory = 'MCU/projects/bms_bringup/Core'
New-Item -ItemType Directory -Force -Path 'build/host' | Out-Null

$nativeCases = @(
    'ov', 'uv', 'timeout', 'pack_voltage', 'temperature',
    'current', 'faults', 'state_machine', 'recovery', 'controller'
)

foreach ($caseName in $nativeCases) {
    $nativeSources = @("$coreDirectory/Src/bms_protection.c")
    if ($caseName -in @('state_machine', 'recovery', 'controller')) {
        $nativeSources += "$coreDirectory/Src/bms_state_machine.c"
    }
    if ($caseName -in @('recovery', 'controller')) {
        $nativeSources += "$coreDirectory/Src/bms_recovery.c"
    }
    if ($caseName -eq 'controller') {
        $nativeSources += "$coreDirectory/Src/bms_controller.c"
    }

    $nativeOutput = "build/host/test_bms_$caseName.exe"
    & $nativeCompiler -std=c99 -Wall -Wextra -Werror `
        -I "$coreDirectory/Inc" @nativeSources `
        "$coreDirectory/tests/test_bms_$caseName.c" -o $nativeOutput
    if ($LASTEXITCODE -ne 0) { throw "编译失败：$caseName" }

    & ".\$nativeOutput"
    if ($LASTEXITCODE -ne 0) { throw "测试失败：$caseName" }
}
```

- & 调用指定路径的程序；PowerShell 中仅写带引号路径不会自动执行。
- -std=c99 指定 C99；-I 指定头文件目录；-o 指定生成的可执行文件。
- -Wall、-Wextra 开启警告，-Werror 使警告也导致构建失败。
- 每组只链接所需业务模块与一个测试 main；多个测试 main 不能一起链接到同一程序。
- assert 检查实际结果是否符合预期，未定义 NDEBUG 时断言有效。退出码非零应保留原始输出并停止处理。
- 电脑测试验证的是给定输入下的 C 逻辑。Keil 构建验证 ARM 工程能编译链接；下载和上板实验另验证外设与实际电路。三者证据不能互相替代。

static void AssertReport(
    BmsVoltageReport report,
    uint8_t expected_ov,
    uint8_t expected_invalid,
    uint8_t expected_timeout,
    uint8_t expected_input_error)
{

    assert(report.invalid_mask == expected_invalid);

{
    BmsSample sample;
    BmsSample empty_sample = {0}; /* 模拟上电后尚未取得采样的快照。 */

    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4200;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x01u, 0u, 0u, 0u);

    /* 3：第2节过压，bit1 应为 1。 */
    sample = MakeNormalSample();
    sample.cell_mv[1].value = 4300;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x02u, 0u, 0u, 0u);

    /* 4：第3节过压，bit2 应为 1。 */
    sample = MakeNormalSample();
    sample.cell_mv[2].value = 4300;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x04u, 0u, 0u, 0u);

    /* 5：第1、3节同时过压，001 | 100 = 101，即十六进制 0x05。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4300;
    sample.cell_mv[2].value = 4300;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x05u, 0u, 0u, 0u);

    /* 6：无效标志、无来源、负电压，分别出现在三节上。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].valid = 0u;
    sample.cell_mv[1].source = BMS_SOURCE_NONE;
    sample.cell_mv[2].value = -1;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0x07u, 0u, 0u);

    /* 7：同时出现过压、无效和超时，不能在发现第一项后提前结束。 */
    sample = MakeNormalSample();
    sample.cell_mv[0].value = 4300;
    sample.cell_mv[1].valid = 0u;
    sample.cell_mv[2].timestamp_ms = 999u; /* 数据年龄刚好 500 ms。 */
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0x01u, 0x02u, 0x04u, 0u);

    /* 8：第2节旧值虽然高于阈值，本次只报告它超时。 */
    sample = MakeNormalSample();
    sample.cell_mv[1].value = 4300;
    sample.cell_mv[1].timestamp_ms = 999u;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0u, 0x02u, 0u);

    /* 9：没有整组采样对象，必须报告输入错误。 */
    AssertReport(Bms_CheckPackVoltages(
        NULL, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0u, 0u, 1u);

    /* 10：过压阈值为 0，不能当作正常检查结果。 */
    sample = MakeNormalSample();
    AssertReport(Bms_CheckPackVoltages(
        &sample, 0, now_ms, max_age_ms),
        0u, 0u, 0u, 1u);

    /* 11：超时参数为 0，也必须报告输入错误。 */
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, 0u),
        0u, 0u, 0u, 1u);

    /* 12：计数回绕后，三节数据年龄都刚好为 500 ms。 */
    sample = MakeNormalSample();
    for (i = 0u; i < BMS_CELL_COUNT; i++)
    {
        sample.cell_mv[i].timestamp_ms = UINT32_MAX - 99u;
    }
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, 400u, max_age_ms),
        0u, 0u, 0x07u, 0u);

    /* 13：全零快照代表尚未采集，三节都应报告数据无效。 */
    sample = empty_sample;
    AssertReport(Bms_CheckPackVoltages(
        &sample, ov_threshold_mv, now_ms, max_age_ms),
        0u, 0x07u, 0u, 0u);

    puts("13 pack voltage scenarios passed.");
    return 0;
}
```

MakeNormalSample() 负责构造软件快照，每个场景重新取得一份，避免前一场景对后一场景产生影响。函数名中的 Make 表示构造，不代表硬件采样。

#include <assert.h>
#include <stdio.h>
#include "bms_protection.h"

int main(void)
{
    BmsMeasurement cell = {0};

    /* 3000 mV 仅用于软件测试，实物电池的保护阈值尚未确定。 */

    cell.value = 3000;
    assert(Bms_CheckCellUnderVoltage(&cell, threshold_mv)
           == BMS_UV_TRIGGERED);

    /* 4：低于阈值，也触发欠压。 */
    cell.value = 2900;
    assert(Bms_CheckCellUnderVoltage(&cell, threshold_mv)
           == BMS_UV_TRIGGERED);

    /* 5：标志有效的零电压，应触发欠压。 */
    cell.value = 0;
    assert(Bms_CheckCellUnderVoltage(&cell, threshold_mv)
           == BMS_UV_TRIGGERED);

    /* 6：同样是零电压，标志无效时应返回数据无效。 */
    cell.valid = 0u;
    assert(Bms_CheckCellUnderVoltage(&cell, threshold_mv)
           == BMS_UV_DATA_INVALID);
    cell.valid = 1u;
    cell.value = 3700;

    /* 7：负电压不是本项目认可的有效单体电压。 */
    cell.value = -1;
    assert(Bms_CheckCellUnderVoltage(&cell, threshold_mv)
           == BMS_UV_DATA_INVALID);
    cell.value = 3700;

    /* 8：没有来源，不能当作有效测量。 */
    cell.source = BMS_SOURCE_NONE;
    assert(Bms_CheckCellUnderVoltage(&cell, threshold_mv)
           == BMS_UV_DATA_INVALID);
    cell.source = BMS_SOURCE_SIMULATED;

    /* 9：零阈值属于参数错误。 */
    assert(Bms_CheckCellUnderVoltage(&cell, 0)
           == BMS_UV_DATA_INVALID);

    /* 10：负阈值同样属于参数错误。 */
    assert(Bms_CheckCellUnderVoltage(&cell, -1)
           == BMS_UV_DATA_INVALID);

    /* 11：空指针应返回无效，不能访问它的成员。 */
    assert(Bms_CheckCellUnderVoltage(NULL, threshold_mv)
           == BMS_UV_DATA_INVALID);

    puts("11 undervoltage tests passed.");
    return 0;
}

        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_INPUT_ERROR)
        == BMS_FAULT_INPUT_ERROR);

    /* 16：没有电压报告，不能认为全部正常。 */
    assert(Bms_CombineFaults(
        NULL, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_INPUT_ERROR);

    /*
     * 17：不同通道同时故障，所有对应位都要保留。
     * 第1节过压、第2节欠压、第3节无效；
     * 温度超时，同时放电过流。
     */
    voltage.over_voltage_mask = 0x01u;
    voltage.under_voltage_mask = 0x02u;
    voltage.invalid_mask = 0x04u;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_TIMEOUT, BMS_CURRENT_DISCHARGE_OVER_LIMIT)
        == (BMS_FAULT_CELL_OV |
            BMS_FAULT_CELL_UV |
            BMS_FAULT_DATA_INVALID |
            BMS_FAULT_SAMPLE_TIMEOUT |
            BMS_FAULT_DISCHARGE_OC));

    /* 18：电压报告为空时，仍保留过温和充电过流。 */
    assert(Bms_CombineFaults(
        NULL, BMS_TEMP_OVER_LIMIT, BMS_CURRENT_CHARGE_OVER_LIMIT)
        == (BMS_FAULT_INPUT_ERROR |
            BMS_FAULT_OVER_TEMP |
            BMS_FAULT_CHARGE_OC));

    /* 19：新的检查全部正常，汇总函数不保留历史故障。 */
    voltage = empty_report;
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, BMS_CURRENT_NORMAL)
        == BMS_FAULT_NONE);

    /* 20：传入未定义的温度结果，报告输入错误。 */
    assert(Bms_CombineFaults(
        &voltage, (BmsTemperatureResult)99, BMS_CURRENT_NORMAL)
        == BMS_FAULT_INPUT_ERROR);

    /* 21：传入未定义的电流结果，报告输入错误。 */
    assert(Bms_CombineFaults(
        &voltage, BMS_TEMP_NORMAL, (BmsCurrentResult)99)
        == BMS_FAULT_INPUT_ERROR);

    puts("21 fault aggregation tests passed.");
    return 0;
}
```

21 项断言覆盖正常、所有单项映射、三类数据无效与超时、输入错误、空指针、多故障并存、连续调用不保留历史及未知枚举值。测试直接构造检查结果，目的是验证汇总映射；原有六组测试继续验证测量检查逻辑。

empty_report 是字段为零的结构体。voltage = empty_report 是结构体赋值，重置全部字段，让后一个场景不受前一个场景的数据影响。这里手填正常结果仅作为测试输入，实际系统必须调用前置检查函数取得结果。

配套操作见[开发步骤](开发步骤.md#s16)。以下是对应学习阶段的测试方法与示例，不作为新增实测结果；最新源码测试使用本文统一回归命令。

### 第三步：新建电脑测试

新建 Core/tests/test_bms_state_machine.c：

```c
#include <assert.h>
#include <stddef.h>
#include <stdio.h>
#include "bms_state_machine.h"

/* 每个场景同时检查状态、锁定故障和负载许可。 */
static void CheckState(
    const BmsStateMachine *machine,
    BmsState expected_state,
    uint32_t expected_faults,
    uint8_t expected_load)
{
    assert(machine->state == expected_state);
    assert(machine->latched_faults == expected_faults);
    assert(machine->load_enabled == expected_load);
}

int main(void)
{
    BmsStateMachine machine;
    BmsControlInput input = {0};

    /* 1：模拟上电，先自检，负载关闭。 */
    Bms_StateMachineInit(&machine);
    CheckState(&machine, BMS_STATE_SELF_TEST, BMS_FAULT_NONE, 0u);

    /* 2：没有故障但自检尚未通过，继续等待。 */
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_SELF_TEST, BMS_FAULT_NONE, 0u);

    /* 3：自检未通过时收到启动指令，也不能运行。 */
    input.command = BMS_CMD_START;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_SELF_TEST, BMS_FAULT_NONE, 0u);

    /* 4：自检通过；同一轮的 START 不直接启动负载。 */
    input.self_test_passed = 1u;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 5：在待机时另发一次 START，进入运行。 */
    input.command = BMS_CMD_START;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_RUN, BMS_FAULT_NONE, 1u);

    /* 6：无新指令、无故障，保持运行。 */
    input.command = BMS_CMD_NONE;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_RUN, BMS_FAULT_NONE, 1u);

    /* 7：停止后回待机，关闭负载许可。 */
    input.command = BMS_CMD_STOP;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 8：待机收到 CLEAR 不会启动。 */
    input.command = BMS_CMD_CLEAR;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 9：启动指令与过压同时出现，故障优先。 */
    input.command = BMS_CMD_START;
    input.active_faults = BMS_FAULT_CELL_OV;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    /* 10：过压消失，但 START 不能解除锁定。 */
    input.active_faults = BMS_FAULT_NONE;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    /* 11：STOP 也不能解除锁定。 */
    input.command = BMS_CMD_STOP;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    /* 12：收到 CLEAR，但恢复条件未满足，保持锁定。 */
    input.command = BMS_CMD_CLEAR;
    input.clear_allowed = 0u;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    /* 13：锁定期间又出现过温，累计记录；故障优先于 CLEAR。 */
    input.clear_allowed = 1u;
    input.active_faults = BMS_FAULT_OVER_TEMP;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK,
        BMS_FAULT_CELL_OV | BMS_FAULT_OVER_TEMP, 0u);

    /* 14：即使允许清除，自检未通过仍不能解除锁定。 */
    input.active_faults = BMS_FAULT_NONE;
    input.self_test_passed = 0u;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK,
        BMS_FAULT_CELL_OV | BMS_FAULT_OVER_TEMP, 0u);

    /* 15：满足条件并收到 CLEAR，只回待机。 */
    input.self_test_passed = 1u;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 16：清除后没有新指令，不会自动运行。 */
    input.command = BMS_CMD_NONE;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 17：重新发送 START，才允许运行。 */
    input.command = BMS_CMD_START;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_RUN, BMS_FAULT_NONE, 1u);

    /* 18：运行中数据无效和超时，故障优先于 STOP。 */
    input.command = BMS_CMD_STOP;
    input.active_faults = BMS_FAULT_DATA_INVALID | BMS_FAULT_SAMPLE_TIMEOUT;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK,
        BMS_FAULT_DATA_INVALID | BMS_FAULT_SAMPLE_TIMEOUT, 0u);

    /* 19：数据恢复、允许清除，CLEAR 返回待机。 */
    input.active_faults = BMS_FAULT_NONE;
    input.command = BMS_CMD_CLEAR;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 20：再次启动，为测试运行中输入错误做准备。 */
    input.command = BMS_CMD_START;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_RUN, BMS_FAULT_NONE, 1u);

    /* 21：输入指针为空，必须撤销负载许可并锁定。 */
    Bms_StateMachineStep(&machine, NULL);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_INPUT_ERROR, 0u);

    /* 22：异常状态值也必须进入锁定。 */
    Bms_StateMachineInit(&machine);
    machine.state = (BmsState)99;
    input.command = BMS_CMD_NONE;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_INPUT_ERROR, 0u);

    /* 23：模拟复位，重新自检，不恢复原先运行状态。 */
    Bms_StateMachineInit(&machine);
    CheckState(&machine, BMS_STATE_SELF_TEST, BMS_FAULT_NONE, 0u);

    /* 24：自检阶段发生数据故障，也要锁定。 */
    input.self_test_passed = 0u;
    input.active_faults = BMS_FAULT_DATA_INVALID;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_DATA_INVALID, 0u);

    /* 25：故障消失，但不能用 CLEAR 绕过尚未通过的自检。 */
    input.active_faults = BMS_FAULT_NONE;
    input.command = BMS_CMD_CLEAR;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_DATA_INVALID, 0u);

    /* 26：自检通过后，允许清除并返回待机。 */
    input.self_test_passed = 1u;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 27：待机时若自检通过标志为0，也拒绝启动。 */
    input.self_test_passed = 0u;
    input.command = BMS_CMD_START;
    Bms_StateMachineStep(&machine, &input);
    CheckState(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    puts("27 state machine scenarios passed.");
    return 0;
}
```

## 恢复区间与稳定计时：测试与验收方法

配套操作见[开发步骤](开发步骤.md#s17)。以下是对应学习阶段的测试方法与示例，不作为新增实测结果；最新源码测试使用本文统一回归命令。

### 第三步：新建电脑测试

新建 Core/tests/test_bms_recovery.c：

```c
#include <assert.h>
#include <stddef.h>
#include <stdio.h>
#include "bms_recovery.h"
#include "bms_state_machine.h"

static unsigned int recovery_checks = 0u;
static unsigned int control_checks = 0u;

/* 模拟本轮取得了一组新的正常采样；仅供测试使用。 */
static void MakeHealthySample(BmsSample *sample, uint32_t now_ms)
{
    unsigned int i;

    for (i = 0u; i < BMS_CELL_COUNT; ++i)
    {
        sample->cell_mv[i].value = 3800;
        sample->cell_mv[i].timestamp_ms = now_ms;
        sample->cell_mv[i].valid = 1u;
        sample->cell_mv[i].source = BMS_SOURCE_SIMULATED;
    }

    sample->temperature_dC.value = 250;
    sample->temperature_dC.timestamp_ms = now_ms;
    sample->temperature_dC.valid = 1u;
    sample->temperature_dC.source = BMS_SOURCE_SIMULATED;

    sample->current_ma.value = 0;
    sample->current_ma.timestamp_ms = now_ms;
    sample->current_ma.valid = 1u;
    sample->current_ma.source = BMS_SOURCE_SIMULATED;
}

/* 为异常测试构造“已经稳定足够久”的计时记录。 */
static void PrepareCase(BmsRecovery *recovery, BmsSample *sample)
{
    MakeHealthySample(sample, 10000u);
    recovery->timing = 1u;
    recovery->stable_since_ms = 8000u;
}

static void CheckRecovery(
    BmsRecovery *recovery,
    const BmsSample *sample,
    const BmsRecoveryConfig *config,
    uint32_t active_faults,
    uint32_t now_ms,
    uint8_t expected_allowed,
    uint8_t expected_timing)
{
    assert(Bms_RecoveryUpdate(
        recovery, sample, config, active_faults, now_ms)
        == expected_allowed);

    if (recovery != NULL)
    {
        assert(recovery->timing == expected_timing);
    }

    ++recovery_checks;
}

static void CheckControl(
    const BmsStateMachine *machine,
    BmsState state,
    uint32_t faults,
    uint8_t enabled)
{
    assert(machine->state == state);
    assert(machine->latched_faults == faults);
    assert(machine->load_enabled == enabled);
    ++control_checks;
}

int main(void)
{
    BmsRecovery recovery;
    BmsSample sample = {0};
    BmsRecoveryConfig config = {
        3200, 4100, 550, 800, 300,
        500u, 500u, 500u, 2000u
    };
    BmsRecoveryConfig bad_config;
    BmsStateMachine machine;
    BmsControlInput input = {0};

    /* 初始化不能把0 ms误认为已经开始计时。 */
    Bms_RecoveryInit(&recovery);
    assert(recovery.timing == 0u);
    assert(recovery.stable_since_ms == 0u);

    /* 从1000 ms开始，2999 ms不足2 s，3000 ms刚好达到。 */
    MakeHealthySample(&sample, 1000u);
    CheckRecovery(&recovery, &sample, &config, 0u, 1000u, 0u, 1u);
    MakeHealthySample(&sample, 2999u);
    CheckRecovery(&recovery, &sample, &config, 0u, 2999u, 0u, 1u);
    MakeHealthySample(&sample, 3000u);
    CheckRecovery(&recovery, &sample, &config, 0u, 3000u, 1u, 1u);
    MakeHealthySample(&sample, 3100u);
    CheckRecovery(&recovery, &sample, &config, 0u, 3100u, 1u, 1u);

    /* 恢复边界以内，仍然允许；电流两个方向分别验证。 */
    PrepareCase(&recovery, &sample);
    sample.cell_mv[0].value = 3201;
    sample.cell_mv[1].value = 4099;
    sample.temperature_dC.value = 549;
    sample.current_ma.value = 799;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 1u, 1u);
    sample.current_ma.value = -299;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 1u, 1u);

    /* 即使测量在恢复区间，有活动故障也必须撤销许可。 */
    PrepareCase(&recovery, &sample);
    CheckRecovery(&recovery, &sample, &config,
        BMS_FAULT_INPUT_ERROR, 10000u, 0u, 0u);

    /* 达到任一恢复边界，都必须清掉计时。 */
    PrepareCase(&recovery, &sample);
    sample.cell_mv[0].value = 4100;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.cell_mv[1].value = 3200;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.temperature_dC.value = 550;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.current_ma.value = 800;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.current_ma.value = -300;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    /* 无效数据与没有来源的数据不能用于恢复。 */
    PrepareCase(&recovery, &sample);
    sample.cell_mv[2].valid = 0u;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.temperature_dC.source = BMS_SOURCE_NONE;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    /* 各类数据年龄达到500 ms，均撤销许可。 */
    PrepareCase(&recovery, &sample);
    sample.cell_mv[0].timestamp_ms = 9500u;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.temperature_dC.timestamp_ms = 9500u;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    sample.current_ma.timestamp_ms = 9500u;
    CheckRecovery(&recovery, &sample, &config, 0u, 10000u, 0u, 0u);

    /* 错误配置不能放行，并清除已有计时。 */
    PrepareCase(&recovery, &sample);
    bad_config = config;
    bad_config.cell_max_mv = bad_config.cell_min_mv;
    CheckRecovery(&recovery, &sample, &bad_config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    bad_config = config;
    bad_config.discharge_limit_ma = 0;
    CheckRecovery(&recovery, &sample, &bad_config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    bad_config = config;
    bad_config.voltage_max_age_ms = 0u;
    CheckRecovery(&recovery, &sample, &bad_config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    bad_config = config;
    bad_config.stable_time_ms = 0u;
    CheckRecovery(&recovery, &sample, &bad_config, 0u, 10000u, 0u, 0u);

    PrepareCase(&recovery, &sample);
    CheckRecovery(&recovery, NULL, &config, 0u, 10000u, 0u, 0u);
    PrepareCase(&recovery, &sample);
    CheckRecovery(&recovery, &sample, NULL, 0u, 10000u, 0u, 0u);
    CheckRecovery(NULL, &sample, &config, 0u, 10000u, 0u, 0u);

    /* 中途异常后，必须重新累计完整的2 s。 */
    Bms_RecoveryInit(&recovery);
    MakeHealthySample(&sample, 1000u);
    CheckRecovery(&recovery, &sample, &config, 0u, 1000u, 0u, 1u);
    MakeHealthySample(&sample, 2500u);
    sample.current_ma.valid = 0u;
    CheckRecovery(&recovery, &sample, &config, 0u, 2500u, 0u, 0u);
    MakeHealthySample(&sample, 2600u);
    CheckRecovery(&recovery, &sample, &config, 0u, 2600u, 0u, 1u);
    MakeHealthySample(&sample, 4599u);
    CheckRecovery(&recovery, &sample, &config, 0u, 4599u, 0u, 1u);
    MakeHealthySample(&sample, 4600u);
    CheckRecovery(&recovery, &sample, &config, 0u, 4600u, 1u, 1u);

    /* 毫秒计数回绕后，仍能判断1999 ms和2000 ms。 */
    Bms_RecoveryInit(&recovery);
    MakeHealthySample(&sample, UINT32_MAX - 999u);
    CheckRecovery(&recovery, &sample, &config,
        0u, UINT32_MAX - 999u, 0u, 1u);
    MakeHealthySample(&sample, 999u);
    CheckRecovery(&recovery, &sample, &config, 0u, 999u, 0u, 1u);
    MakeHealthySample(&sample, 1000u);
    CheckRecovery(&recovery, &sample, &config, 0u, 1000u, 1u, 1u);

    /* 从0 ms开始计时也必须等待完整时间。 */
    Bms_RecoveryInit(&recovery);
    MakeHealthySample(&sample, 0u);
    CheckRecovery(&recovery, &sample, &config, 0u, 0u, 0u, 1u);
    MakeHealthySample(&sample, 1999u);
    CheckRecovery(&recovery, &sample, &config, 0u, 1999u, 0u, 1u);
    MakeHealthySample(&sample, 2000u);
    CheckRecovery(&recovery, &sample, &config, 0u, 2000u, 1u, 1u);

    /* 联合验证：过早的 CLEAR 不会被保存到计时完成后执行。 */
    Bms_StateMachineInit(&machine);
    Bms_RecoveryInit(&recovery);
    input.self_test_passed = 1u;
    input.active_faults = BMS_FAULT_CELL_OV;
    Bms_StateMachineStep(&machine, &input);
    CheckControl(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    input.active_faults = BMS_FAULT_NONE;
    input.command = BMS_CMD_CLEAR;
    MakeHealthySample(&sample, 1000u);
    input.clear_allowed = Bms_RecoveryUpdate(
        &recovery, &sample, &config, input.active_faults, 1000u);
    Bms_StateMachineStep(&machine, &input);
    CheckControl(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    input.command = BMS_CMD_NONE;
    MakeHealthySample(&sample, 3000u);
    input.clear_allowed = Bms_RecoveryUpdate(
        &recovery, &sample, &config, input.active_faults, 3000u);
    assert(input.clear_allowed == 1u);
    Bms_StateMachineStep(&machine, &input);
    CheckControl(&machine, BMS_STATE_FAULT_LOCK, BMS_FAULT_CELL_OV, 0u);

    /* 已有许可，再收到新的 CLEAR 才回到待机。 */
    input.command = BMS_CMD_CLEAR;
    MakeHealthySample(&sample, 3100u);
    input.clear_allowed = Bms_RecoveryUpdate(
        &recovery, &sample, &config, input.active_faults, 3100u);
    Bms_StateMachineStep(&machine, &input);
    CheckControl(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    /* 退出锁定后清掉计时，不能复用到下一次故障。 */
    Bms_RecoveryInit(&recovery);
    input.clear_allowed = 0u;
    input.command = BMS_CMD_NONE;
    Bms_StateMachineStep(&machine, &input);
    CheckControl(&machine, BMS_STATE_STANDBY, BMS_FAULT_NONE, 0u);

    input.command = BMS_CMD_START;
    Bms_StateMachineStep(&machine, &input);
    CheckControl(&machine, BMS_STATE_RUN, BMS_FAULT_NONE, 1u);

    printf("%u recovery checks and %u control scenarios passed.\n",
        recovery_checks, control_checks);
    return 0;
}
```

MakeHealthySample() 表示测试假定本轮取得了新采样，因此填写本轮时间；真实系统只有实际获得新数据时才更新该通道时间戳。测试直接推进虚拟毫秒值，不需要等待真实2秒。

PrepareCase() 构造已经累计足够时间的记录，再注入异常，确保异常确实能撤销原本会成立的许可。CheckRecovery() 检查返回许可和计时标志，自动累计检查次数；CheckControl() 检查状态、锁定故障和软件输出许可。

35 项恢复检查覆盖稳定时间前后及边界、恢复区间内外与相等边界、两个电流方向、活动故障阻止、无效与超时、错误配置、空指针、中断后重计时、时间回绕及0 ms起点。6 个联合场景验证锁定、过早 CLEAR 拒绝、计时完成仍不自动清除、新 CLEAR 返回待机、清除后等待及新 START 运行。

static void MakeHealthySample(BmsSample *sample, uint32_t now_ms)
{
    unsigned int i;

    for (i = 0u; i < BMS_CELL_COUNT; ++i)
    {
        sample->cell_mv[i].value = 3800;
        sample->cell_mv[i].timestamp_ms = now_ms;
        sample->cell_mv[i].valid = 1u;
        sample->cell_mv[i].source = BMS_SOURCE_SIMULATED;
    }

    sample->temperature_dC.value = 250;
    sample->temperature_dC.timestamp_ms = now_ms;
    sample->temperature_dC.valid = 1u;
    sample->temperature_dC.source = BMS_SOURCE_SIMULATED;

    sample->current_ma.value = 0;
    sample->current_ma.timestamp_ms = now_ms;
    sample->current_ma.valid = 1u;
    sample->current_ma.source = BMS_SOURCE_SIMULATED;

static void CheckController(
    const BmsController *controller,
    BmsState state,
    uint32_t active_faults,
    uint32_t latched_faults,
    uint8_t enabled,
    uint8_t clear_allowed)
{
    assert(controller->machine.state == state);
    assert(controller->active_faults == active_faults);
    assert(controller->machine.latched_faults == latched_faults);
    assert(controller->machine.load_enabled == enabled);
    assert(controller->clear_allowed == clear_allowed);
    ++scenarios;
}

    BmsSample sample = {0};

        3000, 4200, 600, 1000, 500,

    };

        BMS_FAULT_CELL_OV | BMS_FAULT_CELL_UV |
        BMS_FAULT_OVER_TEMP | BMS_FAULT_DISCHARGE_OC;

        BMS_CMD_NONE, 0u, 1000u);
    CheckController(&controller, BMS_STATE_SELF_TEST, 0u, 0u, 0u, 0u);

    Bms_ControllerStep(&controller, &sample, &config,

        BMS_CMD_START, 1u, 1200u);

    Bms_ControllerStep(&controller, &sample, &config,
        BMS_CMD_STOP, 1u, 1300u);

    Bms_ControllerStep(&controller, &sample, &config,

    sample.cell_mv[0].value = 4200;

31个场景通过 CheckController() 自动计数，每个场景检查状态、活动故障、锁定故障、软件负载许可及恢复许可五项；另有计时记录断言。

本测试通过测量值、valid和时间戳触发真实检查链，不直接把活动故障或恢复许可手填给状态机。自检结论仍为测试输入，不能写成实际自检已经完成。

关键序列如下：

| 时间/ms | 输入或操作 | 结果 |
| --- | --- | --- |
| 1200 | 新 START | RUN |
| 1500 | 第1节4200 mV且STOP | 过压锁定优先 |
| 1600 | 第1节4150 mV且CLEAR | 活动过压消失，恢复区间未满足，仍锁定 |
| 1700 | 恢复正常且CLEAR | 开始计时，拒绝过早清除 |
| 3699 | 正常，无指令 | 稳定1999 ms，不允许清除 |
| 3700 | 正常，无指令 | 许可为1，但继续锁定 |
| 3800 | 新 CLEAR | 返回待机，清计时 |
| 4000 | 新 START | 重新运行 |

后续序列覆盖多故障并存、无效和超时重新计时，以及错误配置阻止启动。MakeHealthySample() 只表示测试假定取得新采样；真实驱动不能通过刷新旧值的时间戳维持有效。

## HSE 与 72 MHz 系统时钟：测试与验收方法

配套操作见[开发步骤](开发步骤.md#s30)。以下是对应学习阶段的测试方法与示例，不作为新增实测结果；最新源码测试使用本文统一回归命令。

### 编译、下载和运行时验收

#### 构建与时钟变量

1. Keil F7构建，要求0错误0警告，再F8下载并查看校验结果。
2. 在main里的MX_GPIO_Init调用处F9设断点，进入目标调试并F5运行到这里。这个位置在SystemClock_Config之后、GPIO初始化之前。
3. Watch添加SystemCoreClock，应为72000000（若十六进制显示则为0x044AA200）。
4. 若仍停在HAL_Init入口，时钟配置尚未执行，看到8000000并不能判定配置失败；先运行到上述位置。

SystemCoreClock是由软件按硬件状态计算的变量。它支持配置检查，但单独显示72000000不证明实际外部晶体频率或测量精度。

#### RCC寄存器状态

在Peripherals → System Viewer → RCC查看；具体菜单以本机Keil设备视图为准：

| 寄存器字段 | 期望 | 含义 |
| --- | --- | --- |
| RCC_CR.HSERDY | 1 | HSE已就绪 |
| RCC_CR.PLLRDY | 1 | PLL已锁定 |
| RCC_CFGR.PLLSRC | 1 / HSE | PLL参考来自HSE |
| RCC_CFGR.PLLXTPRE | 0 / HSE不分频 | HSE原频率进入PLL |
| RCC_CFGR.PLLMUL | ×9；原始编码0111 | 编码7表示乘9，不能把编码直接当倍频数 |
| RCC_CFGR.SWS | 10（二进制）/2 | 硬件报告当前SYSCLK实际来自PLL |
| RCC_CFGR.PPRE1 | /2 | APB1为36 MHz |
| RCC_CFGR.PPRE2 | /1 | APB2为72 MHz |

SW为软件请求，SWS为硬件实际状态，验收实际时钟来源要看SWS。若寄存器界面不可用，可提供Clock Configuration图及SystemCoreClock值后再指导查看，不为方便读寄存器新增业务代码。

- F5继续执行，移除本次断点后观察约10 s。

- 串口仍为115200/8N1，完整消息仍约每200 ms一次，无乱码。
- 退出调试，断开调试器连线，只保留板卡USB供电，按RESET复测相同功能。RST仍不连失电调试器，避免重现此前已定位的连接影响。

当前.uvoptx仍有-TC10000000，之前States/Sec显示曾对应10 MHz基准。若使用这些时间显示，在Debug → Settings → Trace页核对Core Clock为72 MHz；只校正调试显示/跟踪解释，不改变MCU真实时钟，也不要求启用SWO。
